func isPunctuation1(s string) bool{
	if len(s) != 1{
	return false
}

',':  true,
'.':  true,
'!':  true,
'?':  true,
';':  true,
':':  true,
'-':  true,
'_':  true,
'\'': true,
'"':  true,
}

r:= rune(s[0])
return punctuations[r]
}
